```ts [nuxt.config.ts]
export default defineNuxtConfig({
  modules: ['@lupinum/nuxt-board']
})
```

```vue [app.vue]
<script setup lang="ts">
const engine = createBoardEngine()
engine.createNode({
  type: 'text',
  x: 96,
  y: 96,
  width: 240,
  height: 120,
  data: { content: 'Nuxt auto-imported this API.' }
})
</script>

<template>
  <BoardRoot :engine="engine" class="h-screen" />
</template>
```

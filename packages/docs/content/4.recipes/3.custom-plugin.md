This recipe walks through building an auto-save plugin from scratch — a practical plugin that serializes the board after changes and persists it to `localStorage`.

## The Plugin Interface

A plugin has a `name` and an `install` function that receives the plugin context:

```ts
import type { BoardPlugin, BoardPluginContext } from '@lupinum/board-core'

function autoSavePlugin(options: { key: string; debounceMs?: number }): BoardPlugin {
  return {
    name: 'auto-save',
    install(ctx: BoardPluginContext) {
      // Plugin setup goes here
    },
  }
}
```

## Step 1: Listen for Changes

Use `command:after` to know when the board state changed. Skip ephemeral commands (camera, interactions):

```ts
const skipPrefixes = ['zoom', 'pan', 'begin', 'end', 'update']

ctx.on('command:after', (name) => {
  if (skipPrefixes.some(p => name.startsWith(p))) return
  scheduleSave()
})
```

## Step 2: Debounce Saves

Avoid saving on every keystroke during rapid editing:

```ts
let timer: ReturnType<typeof setTimeout> | null = null
const delay = options.debounceMs ?? 1000

function scheduleSave() {
  if (timer) clearTimeout(timer)
  timer = setTimeout(save, delay)
}
```

## Step 3: Serialize and Persist

```ts
import { jsonCanvasSerializer } from '@lupinum/board-serializer'

function save() {
  const json = jsonCanvasSerializer.export(ctx)
  localStorage.setItem(options.key, json)
  ctx.emit('command:after', 'auto-save:saved', [], 0)
}
```

## Step 4: Expose an API via `extend`

```ts
ctx.extend('autoSave', {
  saveNow: save,
  getLastSaved: () => localStorage.getItem(options.key),
})
```

## Step 5: TypeScript Module Augmentation

Make `engine.ext.autoSave` type-safe:

```ts
declare module '@lupinum/board-core' {
  interface BoardEngineExtensions {
    autoSave: {
      saveNow: () => void
      getLastSaved: () => string | null
    }
  }
}
```

## Complete Plugin

```ts
import type { BoardPlugin, BoardPluginContext } from '@lupinum/board-core'
import { jsonCanvasSerializer } from '@lupinum/board-serializer'

export function autoSavePlugin(options: {
  key: string
  debounceMs?: number
}): BoardPlugin {
  return {
    name: 'auto-save',
    install(ctx: BoardPluginContext) {
      let timer: ReturnType<typeof setTimeout> | null = null
      const delay = options.debounceMs ?? 1000
      const skipPrefixes = ['zoom', 'pan', 'begin', 'end', 'update']

      function save() {
        const json = jsonCanvasSerializer.export(ctx)
        localStorage.setItem(options.key, json)
      }

      function scheduleSave() {
        if (timer) clearTimeout(timer)
        timer = setTimeout(save, delay)
      }

      ctx.on('command:after', (name) => {
        if (skipPrefixes.some(p => name.startsWith(p))) return
        scheduleSave()
      })

      ctx.extend('autoSave', {
        saveNow: save,
        getLastSaved: () => localStorage.getItem(options.key),
      })
    },
  }
}
```

## Usage

```ts
const engine = createBoardEngine({
  plugins: [
    historyPlugin(),
    autoSavePlugin({ key: 'my-board', debounceMs: 2000 }),
  ],
})

// Manual save
engine.ext.autoSave.saveNow()

// Restore on load
const saved = engine.ext.autoSave.getLastSaved()
if (saved) {
  const doc = jsonCanvasSerializer.parse(saved)
  engine.importJSON(JSON.stringify(jsonCanvasSerializer.toSnapshot(doc)), 'replace')
}
```

---

## Key Patterns Used

- **Plugin context** — `ctx` extends `BoardEngine` with `emit()`, `extend()`, `runCommand()`
- **Extension API** — `ctx.extend()` registers typed APIs on `engine.ext`
- **Module augmentation** — TypeScript picks up `engine.ext.autoSave` types
- **Event-driven** — listens to `command:after` instead of polling

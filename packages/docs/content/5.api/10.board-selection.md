## Install

::code-group
```bash [pnpm]
pnpm add @lupinum/board-selection
```
```bash [npm]
npm install @lupinum/board-selection
```
::

This package provides pure utility functions — it is not a plugin and does not need to be installed on the engine.

---

## Functions

### getSelectionNodes

Returns the currently selected nodes as an array.

```ts
function getSelectionNodes(engine: BoardEngine): BoardNode[]
```

```ts
import { getSelectionNodes } from '@lupinum/board-selection'

const selected = getSelectionNodes(engine)
console.log(`${selected.length} nodes selected`)
```

### getSelectionBounds

Returns the combined bounding box of all selected nodes, or `null` if nothing is selected.

```ts
function getSelectionBounds(engine: BoardEngine): Bounds | null
```

```ts
import { getSelectionBounds } from '@lupinum/board-selection'

const bounds = getSelectionBounds(engine)
if (bounds) {
  console.log('Selection spans from', bounds.minX, bounds.minY, 'to', bounds.maxX, bounds.maxY)
}
```

### toggleIds

Toggles node IDs in a selection array. IDs present in both `current` and `ids` are removed; IDs in `ids` but not in `current` are added.

```ts
function toggleIds(current: NodeId[], ids: NodeId[]): NodeId[]
```

```ts
import { toggleIds } from '@lupinum/board-selection'

const updated = toggleIds(['a', 'b'], ['b', 'c'])
// ['a', 'c']
```

## createJsonCanvasSerializer

Creates a new serializer instance. Use this when you need isolated type handler registrations.

```ts
function createJsonCanvasSerializer(): JsonCanvasSerializer
```

## jsonCanvasSerializer

A default shared serializer instance. Convenient for single-board applications.

```ts
const jsonCanvasSerializer: JsonCanvasSerializer
```

## JsonCanvasSerializer API

### registerType

Registers custom serialize/deserialize handlers for a node type. Handlers control how node data is converted to/from JSON Canvas format.

```ts
registerType(type: string, handler: TypeHandler): void
```

```ts
import { jsonCanvasSerializer } from '@lupinum/board-serializer'

jsonCanvasSerializer.registerType('image', {
  serialize: (node) => ({ src: node.data.src, alt: node.data.alt }),
  deserialize: (raw) => ({ src: raw.src as string, alt: raw.alt as string }),
})
```

### export

Exports a board snapshot (or engine) to a JSON Canvas string. Optionally includes edge data.

```ts
export(
  input: BoardSnapshot | BoardEngine,
  extras?: { edges?: Array<Record<string, unknown>> },
): string
```

```ts
const json = jsonCanvasSerializer.export(engine)

// With connections
const edges = engine.ext.connections.getEdges()
const json = jsonCanvasSerializer.export(engine, { edges })
```

### parse

Parses a JSON Canvas string into a document object.

```ts
parse(json: string): JsonCanvasDocument
```

```ts
const document = jsonCanvasSerializer.parse(jsonString)
console.log(document.nodes.length, 'nodes')
```

### toSnapshot

Converts a parsed JSON Canvas document into a `BoardSnapshot` that can be imported by the engine.

```ts
toSnapshot(document: JsonCanvasDocument): BoardSnapshot
```

```ts
const document = jsonCanvasSerializer.parse(jsonString)
const snapshot = jsonCanvasSerializer.toSnapshot(document)
engine.importJSON(JSON.stringify(snapshot), 'replace')
```

### hydrateEngine

Hydrates an engine directly from a parsed JSON Canvas document, including connection edges when the connections plugin is installed.

```ts
hydrateEngine(engine: BoardEngine, document: JsonCanvasDocument, mode?: 'replace' | 'merge'): void
```

```ts
const document = jsonCanvasSerializer.parse(jsonString)
jsonCanvasSerializer.hydrateEngine(engine, document, 'replace')
```

## Round-Trip Example

```ts
import { jsonCanvasSerializer } from '@lupinum/board-serializer'

// Export
const json = jsonCanvasSerializer.export(engine)

// Persist
localStorage.setItem('canvas', json)

// Restore
const stored = localStorage.getItem('canvas')
if (stored) {
  const doc = jsonCanvasSerializer.parse(stored)
  jsonCanvasSerializer.hydrateEngine(engine, doc, 'replace')
}
```

## Types

### JsonCanvasDocument

The top-level JSON Canvas document structure.

```ts
interface JsonCanvasDocument {
  nodes: JsonCanvasNode[]
  edges?: Array<Record<string, unknown>>
  'x-canvas'?: {
    camera?: Camera
    grid?: GridSettings
    nextZIndex?: number
    nodes?: Record<string, {
      zIndex?: number
      locked?: boolean
      visible?: boolean
      parentId?: string
    }>
    edges?: Array<Record<string, unknown>>
  }
}
```

The `x-canvas` extension stores board-specific metadata that is not part of the JSON Canvas spec (camera position, grid settings, node z-order, lock state, group hierarchy, and edge data).

### JsonCanvasNode

A node in the JSON Canvas format.

```ts
interface JsonCanvasNode {
  id: string
  type: string
  x: number
  y: number
  width: number
  height: number
  text?: string
  [key: string]: unknown   // additional fields from type handlers
}
```

### TypeHandler

Custom serialization logic for a node type.

```ts
type TypeHandler = {
  serialize?: (node: BoardNode) => Record<string, unknown>
  deserialize?: (raw: Record<string, unknown>) => Record<string, unknown>
}
```

| Method | Description |
| --- | --- |
| `serialize` | Converts node data to JSON Canvas fields. Called during `export()`. |
| `deserialize` | Converts JSON Canvas fields back to node data. Called during `toSnapshot()`. |

If no handler is registered for a type, the serializer uses the default behavior:
- **Serialize**: stores `data.content` as the `text` field.
- **Deserialize**: reads the `text` field into `data.content`.

### JsonCanvasSerializer

```ts
interface JsonCanvasSerializer {
  registerType(type: string, handler: TypeHandler): void
  export(input: BoardSnapshot | BoardEngine, extras?: { edges?: Array<Record<string, unknown>> }): string
  parse(json: string): JsonCanvasDocument
  toSnapshot(document: JsonCanvasDocument): BoardSnapshot
  hydrateEngine(engine: BoardEngine, document: JsonCanvasDocument, mode?: 'replace' | 'merge'): void
}
```
